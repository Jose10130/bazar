<?php
require_once '../config/database.php';
require_once '../models/Producto.php';

class ProductoController {
    private $db;
    private $producto;

    public function __construct() {
        $database = new Database();
        $this->db = $database->getConnection();
        $this->producto = new Producto($this->db);
    }

    public function listar() {
        return $this->producto->leer();
    }

    public function crear($datos) {
        if ($datos['stock'] < 0) return "El stock no puede ser negativo.";
        
        $this->producto->nombre = $datos['nombre'];
        $this->producto->descripcion = $datos['descripcion'];
        $this->producto->precio = $datos['precio'];
        $this->producto->stock = $datos['stock'];
        $this->producto->minimo_bulto = $datos['minimo_bulto'];
        $this->producto->categoria = $datos['categoria'];
        $this->producto->imagen = $datos['imagen'] ?? 'placeholder.jpg';

        if ($this->producto->crear()) {
            header("Location: admin_productos.php?msg=creado");
        } else {
            return "Error al crear el producto.";
        }
    }

    public function obtenerUno($id) {
        $this->producto->id = $id;
        return $this->producto->leerUno();
    }

    public function editar($id, $datos) {
        if ($datos['stock'] < 0) return "El stock no puede ser negativo.";

        $this->producto->id = $id;
        $this->producto->nombre = $datos['nombre'];
        $this->producto->descripcion = $datos['descripcion'];
        $this->producto->precio = $datos['precio'];
        $this->producto->stock = $datos['stock'];
        $this->producto->minimo_bulto = $datos['minimo_bulto'];
        $this->producto->categoria = $datos['categoria'];

        if ($this->producto->actualizar()) {
            header("Location: admin_productos.php?msg=editado");
        } else {
            return "Error al actualizar el producto.";
        }
    }

    public function eliminar($id) {
        $this->producto->id = $id;
        if ($this->producto->borrar()) {
            header("Location: admin_productos.php?msg=eliminado");
        } else {
            return "Error al eliminar el producto.";
        }
    }
}
?>
