<?php
class Carrito {
    private $conn;
    private $table_name = "carrito";

    public $id;
    public $usuario_id;
    public $producto_id;
    public $cantidad;

    public function __construct($db) {
        $this->conn = $db;
    }

    public function agregar() {
        // Verificar si ya existe para actualizar cantidad o insertar nuevo
        $query_check = "SELECT id, cantidad FROM " . $this->table_name . " WHERE usuario_id = :u AND producto_id = :p";
        $stmt_check = $this->conn->prepare($query_check);
        $stmt_check->bindParam(":u", $this->usuario_id);
        $stmt_check->bindParam(":p", $this->producto_id);
        $stmt_check->execute();
        $row = $stmt_check->fetch(PDO::FETCH_ASSOC);

        if($row) {
            $nueva_cantidad = $row['cantidad'] + $this->cantidad;
            $query = "UPDATE " . $this->table_name . " SET cantidad = :c WHERE id = :id";
            $stmt = $this->conn->prepare($query);
            $stmt->bindParam(":c", $nueva_cantidad);
            $stmt->bindParam(":id", $row['id']);
        } else {
            $query = "INSERT INTO " . $this->table_name . " SET usuario_id=:u, producto_id=:p, cantidad=:c";
            $stmt = $this->conn->prepare($query);
            $stmt->bindParam(":u", $this->usuario_id);
            $stmt->bindParam(":p", $this->producto_id);
            $stmt->bindParam(":c", $this->cantidad);
        }

        return $stmt->execute();
    }

    public function obtenerPorUsuario($usuario_id) {
        $query = "SELECT c.*, p.nombre, p.precio, p.imagen, p.minimo_bulto, p.stock 
                  FROM " . $this->table_name . " c 
                  JOIN productos p ON c.producto_id = p.id 
                  WHERE c.usuario_id = ?";
        $stmt = $this->conn->prepare($query);
        $stmt->execute([$usuario_id]);
        return $stmt;
    }

    public function actualizarCantidad($id, $cantidad) {
        $query = "UPDATE " . $this->table_name . " SET cantidad = :c WHERE id = :id";
        $stmt = $this->conn->prepare($query);
        $stmt->bindParam(":c", $cantidad);
        $stmt->bindParam(":id", $id);
        return $stmt->execute();
    }

    public function eliminar($id) {
        $query = "DELETE FROM " . $this->table_name . " WHERE id = ?";
        $stmt = $this->conn->prepare($query);
        return $stmt->execute([$id]);
    }

    public function vaciar($usuario_id) {
        $query = "DELETE FROM " . $this->table_name . " WHERE usuario_id = ?";
        $stmt = $this->conn->prepare($query);
        return $stmt->execute([$usuario_id]);
    }
}
?>
