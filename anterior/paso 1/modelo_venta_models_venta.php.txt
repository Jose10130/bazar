<?php
class Venta {
    private $conn;
    private $table_venta = "ventas";
    private $table_detalle = "detalle_venta";

    public $id;
    public $usuario_id;
    public $total;
    public $estado;

    public function __construct($db) {
        $this->conn = $db;
    }

    // Crear venta (Transacción)
    public function crear($items) {
        try {
            $this->conn->beginTransaction();

            // 1. Insertar Cabecera
            $query_v = "INSERT INTO " . $this->table_venta . " SET usuario_id=:u, total=:t, estado='pendiente'";
            $stmt_v = $this->conn->prepare($query_v);
            $stmt_v->bindParam(":u", $this->usuario_id);
            $stmt_v->bindParam(":t", $this->total);
            $stmt_v->execute();
            
            $this->id = $this->conn->lastInsertId();

            // 2. Insertar Detalle
            $query_d = "INSERT INTO " . $this->table_detalle . " (venta_id, producto_id, cantidad, precio_unitario) VALUES (?, ?, ?, ?)";
            $stmt_d = $this->conn->prepare($query_d);

            foreach($items as $item) {
                $stmt_d->execute([$this->id, $item['producto_id'], $item['cantidad'], $item['precio']]);
                
                // 3. Descontar Stock
                $query_s = "UPDATE productos SET stock = stock - ? WHERE id = ?";
                $stmt_s = $this->conn->prepare($query_s);
                $stmt_s->execute([$item['cantidad'], $item['producto_id']]);
            }

            $this->conn->commit();
            return true;
        } catch (Exception $e) {
            $this->conn->rollBack();
            return false;
        }
    }

    // Leer ventas
    public function leerTodas() {
        $query = "SELECT v.*, u.nombre as cliente FROM " . $this->table_venta . " v 
                  JOIN usuarios u ON v.usuario_id = u.id ORDER BY v.fecha_venta DESC";
        $stmt = $this->conn->prepare($query);
        $stmt->execute();
        return $stmt;
    }
}
?>
