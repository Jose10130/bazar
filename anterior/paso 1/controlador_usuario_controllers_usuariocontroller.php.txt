<?php
require_once '../config/database.php';
require_once '../models/Usuario.php';

class UsuarioController {
    private $db;
    private $usuario;

    public function __construct() {
        $database = new Database();
        $this->db = $database->getConnection();
        $this->usuario = new Usuario($this->db);
    }

    public function registrar($datos) {
        // Validaciones básicas
        if (!filter_var($datos['email'], FILTER_VALIDATE_EMAIL)) {
            return "Email no válido.";
        }
        if (strlen($datos['password']) < 6) {
            return "La contraseña debe tener al menos 6 caracteres.";
        }

        // Verificar email único
        $query = "SELECT id FROM usuarios WHERE email = ?";
        $stmt = $this->db->prepare($query);
        $stmt->execute([$datos['email']]);
        if ($stmt->fetch()) {
            return "El email ya está registrado.";
        }

        $this->usuario->nombre = $datos['nombre'];
        $this->usuario->email = $datos['email'];
        $this->usuario->password = $datos['password'];
        $this->usuario->rol = 'cliente'; // Por defecto

        if ($this->usuario->crear()) {
            header("Location: login.php?msg=registrado");
        } else {
            return "Error al registrar el usuario.";
        }
    }

    public function login($email, $password) {
        $query = "SELECT * FROM usuarios WHERE email = ?";
        $stmt = $this->db->prepare($query);
        $stmt->execute([$email]);
        $user = $stmt->fetch(PDO::FETCH_ASSOC);

        if ($user && password_verify($password, $user['password'])) {
            session_start();
            $_SESSION['usuario_id'] = $user['id'];
            $_SESSION['nombre'] = $user['nombre'];
            $_SESSION['rol'] = $user['rol'];

            if ($user['rol'] == 'admin') {
                header("Location: admin_dashboard.php");
            } else {
                header("Location: catalogo.php");
            }
            exit;
        } else {
            return "Credenciales incorrectas.";
        }
    }

    public static function verificarSesion() {
        if (session_status() == PHP_SESSION_NONE) session_start();
        if (!isset($_SESSION['usuario_id'])) {
            header("Location: login.php");
            exit;
        }
    }

    public static function verificarAdmin() {
        self::verificarSesion();
        if ($_SESSION['rol'] !== 'admin') {
            header("Location: catalogo.php?error=acceso_denegado");
            exit;
        }
    }

    public function logout() {
        session_start();
        session_destroy();
        header("Location: login.php");
    }
}
?>
